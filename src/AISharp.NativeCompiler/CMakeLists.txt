cmake_minimum_required(VERSION 3.24)
project(AISharpNativeCompiler VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The first native host target is Windows/MSVC; other targets are not qualified.")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/core/outcomes.aisharp" AISHARP_CORE_TEXT)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/outcomes.aisharp" AISHARP_CORE_HASH)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/core/outcomes.aisharp" AISHARP_CORE_HEX HEX)
string(REGEX REPLACE "(..)" "\\1;" AISHARP_CORE_BYTES "${AISHARP_CORE_HEX}")
string(SUBSTRING "${AISHARP_CORE_HEX}" 0 6 AISHARP_CORE_PREFIX)
string(FIND "${AISHARP_CORE_TEXT}" ")AISharpCore\"" AISHARP_CORE_DELIMITER)
string(FIND "${AISHARP_CORE_TEXT}" "\r" AISHARP_CORE_CR)
if(NOT AISHARP_CORE_DELIMITER EQUAL -1 OR "0d" IN_LIST AISHARP_CORE_BYTES OR AISHARP_CORE_PREFIX STREQUAL "efbbbf")
  message(FATAL_ERROR "Core outcome source must use LF UTF-8 without a BOM and cannot contain the embedding delimiter.")
endif()
configure_file(core/native_core_outcomes.hpp.in native-core-expected.hpp @ONLY NEWLINE_STYLE UNIX)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/native-core-expected.hpp" AISHARP_CORE_EXPECTED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native_core_outcomes.hpp" AISHARP_CORE_ACTUAL)
string(REPLACE "\r\n" "\n" AISHARP_CORE_ACTUAL "${AISHARP_CORE_ACTUAL}")
if(NOT AISHARP_CORE_ACTUAL STREQUAL AISHARP_CORE_EXPECTED)
  message(FATAL_ERROR "Embedded core source is stale. Run core/generate.py explicitly.")
endif()
add_executable(aisharp-native compiler.cpp driver.cpp)
target_compile_features(aisharp-native PRIVATE cxx_std_20)
target_compile_definitions(aisharp-native PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(aisharp-native PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
if(MSVC)
  # Deeply nested native sources recurse through parser, binder and drop analysis.
  target_link_options(aisharp-native PRIVATE /STACK:16777216)
endif()
set_property(TARGET aisharp-native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(aisharp-native PRIVATE bcrypt)
